-----------------------
1. A browser-only HTML file must NOT create or set another user's Firebase password directly.
2. This update uses Firebase's password-reset email flow for an employee instead. The employee chooses their own password using the secure Firebase email link.

4. Your existing code stores services in localStorage, so service links are per-browser, not shared across employees.

F. ADD THESE JAVASCRIPT FUNCTIONS
---------------------------------
Add these functions after loadEmployees() or near the other admin functions:

let currentRole = "user";
let cachedUsers = [];

async function loadUsers() {
  const list = document.getElementById("usersList");
  const message = document.getElementById("usersMessage");
  if (!list || !db || !currentUser) return;

  list.textContent = "Loading users…";
  message.textContent = "";

  try {
    // Only the admin should be able to read the profiles collection.
    const roleSnap = await getDoc(doc(db, "roles", currentUser.uid));
    if (!roleSnap.exists() || roleSnap.data().role !== "admin") {
      list.textContent = "Access denied: Admin only.";
      return;
    }

    const snapshot = await getDocs(collection(db, "profiles"));
    cachedUsers = snapshot.docs.map(userDoc => ({
      uid: userDoc.id,
      ...userDoc.data()
    }));

    document.getElementById("usersTotal").textContent = cachedUsers.length;
    document.getElementById("usersApproved").textContent =
      cachedUsers.filter(u => u.status === "approved").length;
    document.getElementById("usersPending").textContent =
      cachedUsers.filter(u => !u.status || u.status === "pending").length;
    document.getElementById("usersSuspended").textContent =
      cachedUsers.filter(u => u.status === "suspended").length;

    renderUsersList();
  } catch (error) {
    console.error("User list error:", error);
    list.textContent = "Could not load users. Check Firestore Security Rules.";
  }
}

function renderUsersList() {
  const list = document.getElementById("usersList");
  const query = (document.getElementById("userSearch")?.value || "").trim().toLowerCase();
  const visible = cachedUsers.filter(user =>
    String(user.displayName || "").toLowerCase().includes(query) ||
    String(user.email || "").toLowerCase().includes(query) ||
    String(user.uid || "").toLowerCase().includes(query)
  );

  if (!visible.length) {
    list.textContent = "No matching users found.";
    return;
  }

  list.innerHTML = "";
  visible.forEach(user => {
    const row = document.createElement("div");
    row.className = "user-row";

    const details = document.createElement("div");
    details.style.minWidth = "220px";
    const name = document.createElement("strong");
    name.textContent = user.displayName || "Employee";
    const info = document.createElement("div");
    info.className = "muted";
    info.textContent = `${user.email || "No email"} · Status: ${user.status || "pending"} · UID: ${user.uid}`;
    details.append(name, info);

    const actions = document.createElement("div");
    actions.className = "user-actions";

    const statusButton = document.createElement("button");
    statusButton.className = "btn btn-secondary";
    statusButton.textContent = user.status === "suspended" ? "Unblock" :
      (user.status === "approved" ? "Block" : "Approve");
    statusButton.addEventListener("click", () => {
      const nextStatus = user.status === "suspended" ? "approved" :
        (user.status === "approved" ? "suspended" : "approved");
      changeEmployeeStatus(user.uid, nextStatus).then(loadUsers);
    });

    const resetButton = document.createElement("button");
    resetButton.className = "btn";
    resetButton.textContent = "Password Reset";
    resetButton.addEventListener("click", async () => {
      const email = String(user.email || "").trim();
      if (!email) {
        document.getElementById("usersMessage").textContent = "This user has no email address.";
        return;
      }
      try {
        await sendPasswordResetEmail(auth, email);
        document.getElementById("usersMessage").textContent =
          "Password-reset email requested for " + email + ". The employee must use the email link to choose a new password.";
        document.getElementById("usersMessage").className = "message success";
      } catch (error) {
        console.error(error);
        document.getElementById("usersMessage").textContent =
          "Could not request password reset. Check Firebase Auth settings and try again.";
        document.getElementById("usersMessage").className = "message error";
      }
    });

    actions.append(statusButton, resetButton);
    row.append(details, actions);
    list.appendChild(row);
  });
}

function renderAccount() {
  if (!currentUser) return;
  document.getElementById("accountEmail").textContent = currentUser.email || "—";
  document.getElementById("accountUid").textContent = currentUser.uid || "—";
  document.getElementById("accountRole").textContent = currentRole;
}

document.getElementById("userSearch")?.addEventListener("input", renderUsersList);
document.getElementById("refreshUsersButton")?.addEventListener("click", loadUsers);

document.getElementById("myPasswordResetButton")?.addEventListener("click", async () => {
  const message = document.getElementById("accountMessage");
  try {
    if (!currentUser?.email) throw new Error("No email address found for this account.");
    await sendPasswordResetEmail(auth, currentUser.email);
    message.className = "message success";
    message.textContent = "Password-reset email requested. Check your inbox.";
  } catch (error) {
    message.className = "message error";
    message.textContent = error.message || "Could not send reset email.";
  }
});

For normal employees, use role="user" (or no roles document if your app checks profiles for regular users). Never let users choose their own role in the registration form.

H. IMPORTANT FIRESTORE RULES
----------------------------
The dashboard code alone cannot securely protect employee records. Use Firestore Security Rules that:
- allow a signed-in user to read their own profile;
- allow a user to create only their own profile with status "pending";
- allow only an admin to read all profiles and change employee status;
- prevent ordinary users from writing documents in the roles collection;
- allow only an admin to manage role documents.

Do not use open rules such as `allow read, write: if true;`.
Do not paste service-account private keys into HTML.
For a stronger setup, create/disable Firebase Authentication users through a trusted Cloud Function using the Firebase Admin SDK. This is required if you want the Admin to directly create an account and assign an initial password. The browser-only approach above safely sends a reset email instead.
